uint64 sys_clone(void) {
  uint64 fcn, arg, stack;
  if(argaddr(0, &fcn) < 0 || argaddr(1, &arg) < 0 || argaddr(2, &stack) < 0) return -1;
  struct proc *p = myproc();
  struct proc *np = allocproc();
  if(!np) return -1;

  np->pagetable = p->pagetable; // SHARE Memory!
  np->sz = p->sz;
  np->is_thread = 1;            // Mark as thread
  
  *(np->trapframe) = *(p->trapframe);
  np->trapframe->a0 = arg;
  np->trapframe->sp = stack + 4096; // Top of stack page
  np->trapframe->epc = fcn;

  for(int i=0; i<NOFILE; i++) if(p->ofile[i]) np->ofile[i] = filedup(p->ofile[i]);
  np->cwd = idup(p->cwd);
  np->parent = p;
  
  acquire(&np->lock);
  np->state = RUNNABLE;
  release(&np->lock);
  return np->pid;
}
uint64 sys_join(void) {
  struct proc *p = myproc();
  int havekids, pid;
  struct proc *np;
  acquire(&wait_lock);
  for(;;){
    havekids = 0;
    for(np = proc; np < &proc[NPROC]; np++){
      if(np->parent == p && np->is_thread){
        acquire(&np->lock);
        havekids = 1;
        if(np->state == ZOMBIE){
          pid = np->pid;
          freeproc(np);
          release(&np->lock); release(&wait_lock);
          return pid;
        } release(&np->lock);
      }
    }
    if(!havekids || p->killed){ release(&wait_lock); return -1; }
    sleep(p, &wait_lock);
  }
}

int counter = 0;
void thr_func(void *arg) {
    for(int i=0; i<100000; i++) {
        sem_wait(0); counter++; sem_post(0); // Protected!

}
int main() {
    sem_init(0, 1);
    void *stack1 = malloc(4096); void *stack2 = malloc(4096);
    clone(thr_func, 0, stack1); clone(thr_func, 0, stack2);
    join(); join();
    printf("Final Counter: %d\n", counter); // Exactly 200000
    exit(0);
}
