QUESTION 1: VIRTUAL TO PHYSICAL TRANSLATION (va2pa)
---------------------------------------------------------
1. Create `kernel/vm.c` helper function:
(Add this at the bottom of `vm.c` so it can access `walkpgdir` and `myproc()`)
```c
int sys_va2pa(void) {
  int va;
  if(argint(0, &va) < 0) return -1;
  
  struct proc *p = myproc();
  pte_t *pte = walkpgdir(p->pgdir, (void*)va, 0);
  
  if(!pte || !(*pte & PTE_P) || !(*pte & PTE_U)) return -1;
  
  int pa = PTE_ADDR(*pte) | (va & 0xFFF);
  
  cprintf("PD Index: %d, PT Index: %d, Offset: %d, Flags: P=%d W=%d U=%d\n",
          PDX(va), PTX(va), va & 0xFFF,
          (*pte & PTE_P) ? 1 : 0, (*pte & PTE_W) ? 1 : 0, (*pte & PTE_U) ? 1 : 0);
          
  return pa;
}
```

QUESTION 3: DUMPING THE PAGE TABLE
---------------------------------------------------------
1. `kernel/vm.c`: Add `sys_vmprint`
```c
int sys_vmprint(void) {
  struct proc *p = myproc();
  pde_t *pgdir = p->pgdir;
  int kernel_entries = 0;
  
  cprintf("Page Table for PID %d:\n", p->pid);
  for(int i = 0; i < 512; i++) {
    if(pgdir[i] & PTE_P) {
      if(i >= PDX(KERNBASE)) { kernel_entries++; continue; }
      
      cprintf("PD Index %d: PA 0x%x, Flags P=%d W=%d U=%d\n", i, PTE_ADDR(pgdir[i]), 
              (pgdir[i]&PTE_P)?1:0, (pgdir[i]&PTE_W)?1:0, (pgdir[i]&PTE_U)?1:0);
              
      pte_t *pt = (pte_t*)P2V(PTE_ADDR(pgdir[i]));
      for(int j = 0; j < 512; j++) {
        if(pt[j] & PTE_P) {
          uint va = (i << 22) | (j << 12);
          cprintf("  ..PT Index %d: VA 0x%x -> PA 0x%x, Flags P=%d W=%d U=%d\n", j, va, PTE_ADDR(pt[j]),
                  (pt[j]&PTE_P)?1:0, (pt[j]&PTE_W)?1:0, (pt[j]&PTE_U)?1:0);
        }
      }
    }
  }
  cprintf("Kernel PT entries (>=KERNBASE): %d\n", kernel_entries);
  return 0;
}
```

QUESTION 4: LAZY MEMORY ALLOCATION
---------------------------------------------------------
1. `kernel/sysproc.c`: Modify `sys_sbrk`
```c
int sys_sbrk(void) {
  int addr;
  int n;
  if(argint(0, &n) < 0) return -1;
  addr = myproc()->sz;
  
  if(n > 0) {
    myproc()->sz += n; // Lazy allocation
  } else if (n < 0) {
    myproc()->sz = deallocuvm(myproc()->pgdir, addr, addr + n);
  }
  return addr;
}
```

2. `kernel/trap.c`: Handle Page Fault (T_PGFLT = 14)
Include `mmu.h` at the top if missing.
Inside `trap(struct trapframe *tf)`, add a case for `T_PGFLT`:
```c
  case T_PGFLT: {
    uint fault_addr = rcr2();
    struct proc *p = myproc();
    
    // Check if within heap
    if(fault_addr < p->sz && fault_addr >= (uint)p->tf->esp) {
      char *mem = kalloc();
      if(mem == 0) {
        cprintf("Lazy alloc out of memory\n");
        p->killed = 1;
        break;
      }
      memset(mem, 0, PGSIZE);
      uint a = PGROUNDDOWN(fault_addr);
      if(mappages(p->pgdir, (char*)a, PGSIZE, V2P(mem), PTE_W|PTE_U) < 0) {
        kfree(mem);
        p->killed = 1;
      }
    } else {
      cprintf("Segmentation Fault PID %d VA 0x%x\n", p->pid, fault_addr);
      p->killed = 1;
    }
    break;
  }
```
