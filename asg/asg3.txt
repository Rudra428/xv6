---------------------------------------------------------
1. kernel/proc.c: Replace `for(;;)` in `scheduler()` with:
  for(;;){
    intr_on();
    struct proc *min_p = 0;
    for(p = proc; p < &proc[NPROC]; p++) {
      acquire(&p->lock);
      if(p->state == RUNNABLE) {
        if(!min_p || p->ctime < min_p->ctime) {
          if(min_p) release(&min_p->lock);
          min_p = p;
          continue;
        }
      }
      release(&p->lock);
    }
    if(min_p) {
      min_p->state = RUNNING;
      c->proc = min_p;
      swtch(&c->context, &min_p->context);
      c->proc = 0;
      release(&min_p->lock);
    }
  }

QUESTION 3: SHORTEST JOB FIRST (SJF)
---------------------------------------------------------
1. kernel/proc.c: Replace `for(;;)` in `scheduler()` with:
  for(;;){
    intr_on();
    struct proc *min_p = 0;
    for(p = proc; p < &proc[NPROC]; p++) {
      acquire(&p->lock);
      if(p->state == RUNNABLE) {
        if(!min_p) { min_p = p; continue; }
        // Treat 0 or -1 as lowest priority (run last)
        int bt_p = (p->burst_time > 0) ? p->burst_time : 999999;
        int bt_min = (min_p->burst_time > 0) ? min_p->burst_time : 999999;
        if(bt_p < bt_min) {
          release(&min_p->lock);
          min_p = p; continue;
        }
      }
      release(&p->lock);
    }
    if(min_p) {
      min_p->state = RUNNING;
      c->proc = min_p;
      swtch(&c->context, &min_p->context);
      c->proc = 0;
      release(&min_p->lock);
    }
  }

2. kernel/trap.c: Disable preemption (comment out `yield()` on timer).

3. user/sjftest.c:
#include "kernel/types.h"
#include "user/user.h"
int main() {
    for(int i=4; i>0; i--) { // Burst times: 40, 30, 20, 10
        int pid = fork();
        if(pid == 0) {
            set_burst(getpid(), i * 10);
            for(volatile int j=0; j<50000000*i; j++);
            printf("Process %d (Burst %d) finished\n", getpid(), i*10);
            exit(0);
        }
    }
    for(int i=0; i<4; i++) wait(0);
    exit(0);
}
