QUESTION 2: VIRTUAL TO PHYSICAL (vtop)
---------------------------------------------------------
This is identical to `va2pa` in Assignment 8.
1. `kernel/vm.c`:
```c
int sys_vtop(void) {
  int va;
  if(argint(0, &va) < 0) return -1;
  struct proc *p = myproc();
  pte_t *pte = walkpgdir(p->pgdir, (void*)va, 0);
  if(!pte || !(*pte & PTE_P)) return -1;
  return PTE_ADDR(*pte) | (va & 0xFFF);
}
```

QUESTION 3: PAGE PROTECTION (pageinfo & setreadonly)
---------------------------------------------------------
1. `kernel/vm.c`: Implement `sys_pageinfo` and `sys_setreadonly`
```c
int sys_pageinfo(void) {
  int va;
  if(argint(0, &va) < 0) return -1;
  
  pte_t *pte = walkpgdir(myproc()->pgdir, (void*)va, 0);
  if(!pte || !(*pte & PTE_P)) {
    cprintf("Address 0x%x is not mapped.\n", va);
    return -1;
  }
  
  cprintf("Frame PA: 0x%x, Flags: P=%d W=%d U=%d\n", 
          PTE_ADDR(*pte), (*pte & PTE_P)?1:0, (*pte & PTE_W)?1:0, (*pte & PTE_U)?1:0);
  return 0;
}

int sys_setreadonly(void) {
  int va;
  if(argint(0, &va) < 0) return -1;
  
  pte_t *pte = walkpgdir(myproc()->pgdir, (void*)va, 0);
  if(!pte || !(*pte & PTE_P)) return -1;
  
  *pte &= ~PTE_W; // Clear write permission bit
  
  // Flush TLB so the processor sees the updated page table
  lcr3(V2P(myproc()->pgdir));
  return 0;
}
```

#include "types.h"

// Helper for Optimal to find farthest future use
int get_farthest(int *mem, int frames, int *ref, int ref_len, int current_idx) {
    int farthest_idx = -1;
    int max_dist = -1;
    
    for(int i=0; i<frames; i++) {
        int dist = 9999;
        for(int j=current_idx+1; j<ref_len; j++) {
            if(mem[i] == ref[j]) { dist = j; break; }
        }
        if(dist > max_dist) { max_dist = dist; farthest_idx = i; }
    }
    return farthest_idx;
}
