3. kernel/proc.c (scheduler):
  for(;;){
    intr_on();
    struct proc *min_p = 0;
    for(p = proc; p < &proc[NPROC]; p++) {
      acquire(&p->lock);
      if(p->state == RUNNABLE && p->remaining_time > 0) {
        if(!min_p || p->remaining_time < min_p->remaining_time) {
          if(min_p) release(&min_p->lock);
          min_p = p; continue;
        }
      }
      release(&p->lock);
    }
    if(min_p) {
      min_p->state = RUNNING;
      c->proc = min_p;
      swtch(&c->context, &min_p->context);
      c->proc = 0;
      release(&min_p->lock);
    }
  }

    for(p = proc; p < &proc[NPROC]; p++) {

  }
